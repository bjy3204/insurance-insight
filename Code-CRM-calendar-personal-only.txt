const CUSTOMER_FIELDS = {
  C6:2, F6:3, I6:4, C8:5, C10:6, F8:7, C12:8,
  C16:10, F16:11, I16:12, C18:13, F18:14, I18:15,
  C22:17, F22:18, I22:19, C24:20,
  C28:22, F28:23, I28:24, C30:25, F30:26, I30:27, I8:28
};

function onOpen(e) {
  refreshCrmCalendar();
  refreshMobileRegistrationNotes_();
  SpreadsheetApp.getUi().createMenu('고객 관리')
    .addItem('고객등록', 'openCustomerRegistration')
    .addItem('계약추가', 'activateContractInput')
    .addToUi();
  refreshCustomerCard();
  refreshAllContractCustomerInfo();
  refreshCustomerListLayout();
  refreshOperationalViews();
}

function onEdit(e) {
  refreshMobileRegistrationNotes_(e && e.source);
  if(handleCrmCalendarEdit(e))return;
  if(handleContractEdit(e))return;
  if(e && e.range && crmSheetRole_(e.range.getSheet())==='고객 관리 카드') {
    refreshCustomerCard(e.source);
    return;
  }
  if(e && e.range && crmSheetRole_(e.range.getSheet())==='고객 등록' && e.range.getA1Notation()==='E32' && e.value==='TRUE') {
    try { registerCustomer(e.source); }
    finally { e.range.setValue(false); }
    return;
  }
  handleContactFormatEdit(e);
  handleRenewalEdit(e);
  handleCustomerEdit(e);
  if(e && e.range && crmSheetRole_(e.range.getSheet())==='고객 LIST'){
    refreshCustomerListLayout(e.source);
    refreshCustomerCard(e.source);
    refreshAllContractCustomerInfo(e.source);
    refreshOperationalViews(e.source);
    refreshCrmCalendar(e.source);
  }
}

function selectCustomerCardRows(rows,name,birth) {
  const key=String(name).trim();
  const dob=String(birth).replace(/\D/g,'').slice(-6).padStart(6,'0');
  if(!key)return [];
  return rows.map((row,index)=>({row,index})).filter(item=>
    String(item.row[2]).trim()===key && (!birth || String(item.row[3]).replace(/\D/g,'').slice(0,6)===dob));
}

function refreshCustomerCard(source) {
  const ss=source && typeof source.getSheetByName==='function'?source:SpreadsheetApp.getActiveSpreadsheet();
  const card=crmSheet_(ss,'고객 관리 카드'),list=crmSheet_(ss,'고객 LIST');
  if(!card||!list)return;
  clearCustomerCardContracts_(card);
  const mapping={E6:2,C8:4,C9:5,E10:28,C10:7,C11:6,C12:8,H8:22,J8:23,H9:25,J9:24,H10:26,J10:27,C15:10,E15:11,H15:17,J15:18,C16:13,E16:12,H16:19,J16:20,C17:14,E17:15};
  const targets=Object.keys(mapping).concat(['B1','E5','E8','E9','J4','H4','AH1','AH2','AH3']);
  card.getRangeList(targets).clearContent();
  const name=card.getRange('C4').getDisplayValue().trim();
  if(!name)return;
  const birth=card.getRange('C5').getDisplayValue().trim();
  const range=list.getRange(6,1,Math.max(1,list.getLastRow()-5),28);
  const display=range.getDisplayValues(),raw=range.getValues();
  const found=selectCustomerCardRows(display,name,birth);
  card.getRange('AH2').setValue(found.length);
  if(found.length!==1){
    card.getRange('B1').setValue(found.length===0?'검색 결과가 없습니다. 이름·생년월일을 확인하세요.':birth?'이름과 생년월일이 같은 고객이 여러 명입니다. 고객 LIST를 확인하세요.':'동명이인이 있습니다. 생년월일도 입력하세요.');
    return;
  }
  const item=found[0],row=item.row;
  refreshCustomerCardContracts_(ss,card,name,row[3]);
  Object.entries(mapping).forEach(([address,col])=>{
    const cell=card.getRange(address);
    if(address==='J9')cell.setNumberFormat('m"월 "d"일"').setValue(raw[item.index][col-1]);
    else if(address==='C15')cell.setNumberFormat('0"일"').setValue(raw[item.index][col-1]);
    else cell.setNumberFormat('@').setValue(literalText(row[col-1]));
  });
  card.getRange('AH3').setValue(item.index+6);
  const digits=row[3].replace(/\D/g,'');
  if(!birthFromResidentNumber(digits))return;
  const todayText=Utilities.formatDate(new Date(),ss.getSpreadsheetTimeZone(),'yyyy-MM-dd');
  const parts=todayText.split('-').map(Number),today=new Date(Date.UTC(parts[0],parts[1]-1,parts[2]));
  const yy=Number(digits.slice(0,2)),month=Number(digits.slice(2,4)),day=Number(digits.slice(4,6));
  const century=digits.length===6?(yy<=parts[0]%100?2000:1900):'3478'.includes(digits[6])?2000:'1256'.includes(digits[6])?1900:1800;
  const year=century+yy;
  const age=parts[0]-year-((parts[1]<month||(parts[1]===month&&parts[2]<day))?1:0);
  card.getRange('E8').setValue(age).setNumberFormat('0"세"');
  card.getRange('E9').setValue(digits.length===13?(/[13579]/.test(digits[6])?'남':'여'):'');
  card.getRange('J8').setNumberFormat('@');
  card.getRange('J9').setNumberFormat('m"월 "d"일"');
  card.getRange('C15').setNumberFormat('0"일"');
  function shifted(y,m,d,offset){const base=new Date(Date.UTC(y,m-1+offset,1));const last=new Date(Date.UTC(base.getUTCFullYear(),base.getUTCMonth()+1,0)).getUTCDate();base.setUTCDate(Math.min(d,last));return base;}
  const base=shifted(year,month,day,6);
  let upper=new Date(Date.UTC(parts[0],base.getUTCMonth(),base.getUTCDate()));
  if(upper<today)upper=new Date(Date.UTC(parts[0]+1,base.getUTCMonth(),base.getUTCDate()));
  const serial=Math.round((upper.getTime()-Date.UTC(1899,11,30))/86400000),distance=Math.round((upper-today)/86400000);
  card.getRange('J4').setValue(serial).setNumberFormat('yyyy-mm-dd');
  card.getRange('H4').setValue(distance===0?'D-Day':'D-'+distance);
  card.getRange('AH1').setValue(age+(today>=shifted(year+age,month,day,6)?1:0));
}

function formatPhoneNumber(value) {
  const original=String(value).trim();
  if(!/^[0-9\s-]+$/.test(original))return original;
  const s=original.replace(/\D/g,'');
  if(/^02\d{7,8}$/.test(s))return s.slice(0,2)+'-'+s.slice(2,-4)+'-'+s.slice(-4);
  if(/^050[0-9]\d{8}$/.test(s))return s.slice(0,4)+'-'+s.slice(4,8)+'-'+s.slice(8);
  if(/^0\d{9,10}$/.test(s))return s.slice(0,3)+'-'+s.slice(3,-4)+'-'+s.slice(-4);
  if(/^1\d{7}$/.test(s))return s.slice(0,4)+'-'+s.slice(4);
  return original;
}

function formatCardExpiry(value) {
  const s=String(value).trim();
  if(!s)return '';
  if(!/^\d{2}\/?(0[1-9]|1[0-2])$/.test(s))throw new Error('유효기간은 YYMM 숫자 4자리로 입력하세요. 예: 2801 → 28/01');
  const digits=s.replace('/','');
  return digits.slice(0,2)+'/'+digits.slice(2);
}

function handleContactFormatEdit(e) {
  if(!e||!e.range)return;
  const sheet=e.range.getSheet(),name=crmSheetRole_(sheet);
  const configs=name==='고객 등록'?[{row:8,col:3,fn:formatPhoneNumber},{row:18,col:9,fn:formatCardExpiry}]:name==='고객 LIST'?[{row:null,col:5,fn:formatPhoneNumber},{row:null,col:15,fn:formatCardExpiry}]:[];
  configs.forEach(config=>{
    if(e.range.getColumn()>config.col||e.range.getLastColumn()<config.col)return;
    if(config.row!==null&&(e.range.getRow()>config.row||e.range.getLastRow()<config.row))return;
    const start=config.row===null?Math.max(6,e.range.getRow()):config.row,end=config.row===null?e.range.getLastRow():config.row;
    if(start>end)return;
    const range=sheet.getRange(start,config.col,end-start+1,1),values=range.getDisplayValues(),formulas=range.getFormulas();
    values.forEach((row,i)=>{
      if(!row[0]||formulas[i][0])return;
      const cell=sheet.getRange(start+i,config.col);
      try{cell.setNumberFormat('@').setValue(config.fn(row[0])).clearNote();}
      catch(error){cell.setNote(error.message);if(name==='고객 등록')sheet.getRange('B1').setValue(error.message);else e.source.toast(error.message,'유효기간',8);}
    });
  });
}

function handleRenewalEdit(e) {
  if (!e || !e.range) return;
  const sheet=e.range.getSheet(),name=crmSheetRole_(sheet);
  let column,start,end;
  if(name==='고객 등록' && e.range.getRow()<=28 && e.range.getLastRow()>=28 && e.range.getColumn()<=9 && e.range.getLastColumn()>=9){column=9;start=end=28;}
  else if(name==='고객 LIST' && e.range.getColumn()<=24 && e.range.getLastColumn()>=24 && e.range.getLastRow()>=6){column=24;start=Math.max(6,e.range.getRow());end=e.range.getLastRow();}
  else return;
  const range=sheet.getRange(start,column,end-start+1,1);
  const values=range.getValues(),formulas=range.getFormulas();
  values.forEach((row,i)=>{
    if(formulas[i][0] || row[0]==='' || row[0]===null)return;
    const cell=sheet.getRange(start+i,column);
    try {
      const value=row[0] instanceof Date
        ? Utilities.formatDate(row[0],e.source.getSpreadsheetTimeZone(),'yyyy-MM-dd')
        : row[0];
      cell.setValue(parseRenewalDate(value)).setNumberFormat('m"월 "d"일"');
      if(name==='고객 등록')cell.setNote('갱신일의 월·일을 확인하세요.\n예시 : 260901, 20260901, 26-09-01');
      else cell.clearNote();
    } catch(error) {
      cell.setNumberFormat('@').setNote(name==='고객 등록'
        ? error.message+'\n예시 : 260901, 20260901, 26-09-01'
        : error.message);
      if(name==='고객 등록')sheet.getRange('B1').setValue(error.message);
      else e.source.toast(error.message,'갱신일',8);
    }
  });
}

function handleCustomerEdit(e) {
  if (!e || !e.range) return;
  const sheet = e.range.getSheet();
  if (crmSheetRole_(sheet) === '고객 등록' && e.range.getRow() <= 6 && e.range.getLastRow() >= 6 && e.range.getColumn() <= 9 && e.range.getLastColumn() >= 9) {
    const cell = sheet.getRange('I6');
    if (!cell.getFormula()) {
      const current = cell.getDisplayValue();
      const formatted = formatResidentNumber(current);
      if (formatted !== current) cell.setNumberFormat('@').setValue(formatted);
    }
    return;
  }
  if (crmSheetRole_(sheet) !== '고객 LIST' || e.range.getLastRow() < 6) return;
  // 이름/주민번호/카드 및 청구 생년월일 입력·붙여넣기·지우기 모두 지원.
  const startCol = e.range.getColumn(), endCol = e.range.getLastColumn();
  if (![3,4,12,18].some(col => startCol <= col && col <= endCol)) return;
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(5000)) {
    e.source.toast('다른 등록 작업 중입니다. 주민번호 셀을 다시 편집해 주세요.', '고객 관리');
    return;
  }
  try {
    const start = Math.max(6, e.range.getRow()), end = e.range.getLastRow();
    const values = sheet.getRange(start,3,end-start+1,16).getDisplayValues();
    const formulas = sheet.getRange(start,3,end-start+1,16).getFormulas();
    values.forEach((row,i) => {
      if (startCol <= 4 && endCol >= 4 && !formulas[i][1]) {
        const formatted = formatResidentNumber(row[1]);
        if (formatted !== row[1]) sheet.getRange(start+i,4).setNumberFormat('@').setValue(formatted);
      }
      if (!row[0].trim()) return;
      const dob = birthFromResidentNumber(row[1]);
      if (!dob) return;
      [12,18].forEach(col => {
        const idx = col-3;
        // 직접 입력한 값이나 기존 수식을 덮어쓰지 않습니다.
        if (!row[idx].trim() && !formulas[i][idx])
          sheet.getRange(start+i,col).setNumberFormat('@').setValue(dob);
      });
    });
  } finally { lock.releaseLock(); }
}

function setupCustomerRegistration() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const form = crmSheet_(ss,'고객 등록');
  if (!form || !crmSheet_(ss,'고객 LIST')) throw new Error('고객등록 또는 고객 LIST 시트를 찾지 못했습니다.');
  setupCustomerButton();
  const list = crmSheet_(ss,'고객 LIST');
  if (list.getLastRow() >= 6) {
    handleCustomerEdit({source:ss,range:list.getRange(6,3,list.getLastRow()-5,16)});
  }
  form.getRange('I6').setNumberFormat('@');
  list.getRange(6,4,list.getMaxRows()-5,1).setNumberFormat('@');
  form.getRangeList(['B1','K4','K6','B36']).clearContent();
  ss.toast('등록 기능 연결 완료', '고객 관리');
}

function registerCustomer(source) {
  const ss = source && typeof source.getSheetByName === 'function'
    ? source : SpreadsheetApp.getActiveSpreadsheet();
  const form = crmSheet_(ss,'고객 등록');
  const list = crmSheet_(ss,'고객 LIST');
  if (!form || !list) throw new Error('고객등록 또는 고객 LIST 시트를 찾지 못했습니다.');
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(5000)) {
    form.getRange('B1').setValue('다른 등록 작업 중입니다. 잠시 후 다시 등록하세요.');
    return;
  }
  let committed = false;
  try {
    form.getRangeList(['B1','K4','K6','B36']).clearContent();
    const grid = form.getRange('A1:J32').getDisplayValues();
    const input = {};
    Object.keys(CUSTOMER_FIELDS).forEach(a => {
      const m = /^([A-Z])(\d+)$/.exec(a);
      input[a] = grid[Number(m[2])-1][m[1].charCodeAt(0)-65].trim();
    });
    const digits = input.I6.replace(/\D/g,'');
    const missingName = !input.F6;
    const invalidBirth = ![6,13].includes(digits.length) || !birthFromResidentNumber(digits);
    if (missingName) throw new Error('고객 이름을 입력하세요.');
    if (invalidBirth) throw new Error('주민번호와 생년월일을 확인하세요.');
    input.I6 = digits.length === 6 ? digits : digits.slice(0,6)+'-'+digits.slice(6);
    const dob = birthFromResidentNumber(digits);
    input.F16 = input.F16 || input.F6;
    input.C22 = input.C22 || input.F6;
    input.I16 = input.I16 ? normalizeBirth(input.I16) : dob;
    input.F22 = input.F22 ? normalizeBirth(input.F22) : dob;
    input.C8 = formatPhoneNumber(input.C8);
    input.I18 = formatCardExpiry(input.I18);
    const last = Math.max(6,list.getLastRow());
    const existing = list.getRange(6,3,last-5,2).getDisplayValues();
    if (existing.some(row => row[0].trim() === input.F6 && row[1].replace(/\D/g,'').slice(0,6) === dob))
      throw new Error('같은 이름·생년월일의 고객이 있습니다. 고객 LIST에서 확인하고 수정하세요.');
    let rowNumber = 6;
    existing.forEach((row,i) => { if (row.some(value => value.trim())) rowNumber = i+7; });
    // 다른 열에만 입력된 미완성 행도 덮어쓰지 않습니다.
    while (rowNumber <= list.getMaxRows() && list.getRange(rowNumber,2,1,27).getDisplayValues()[0].some(v => v.trim())) rowNumber++;
    if (rowNumber > list.getMaxRows()) list.insertRowsAfter(list.getMaxRows(),rowNumber-list.getMaxRows());
    const output = Array(27).fill('');
    Object.entries(CUSTOMER_FIELDS).forEach(([address,col]) => {
      output[col-2] = literalText(input[address]);
    });
    if (input.C16) {
      const day = parseCustomerPayday(input.C16);
      output[8] = day;
    }
    if (input.I28) {
      const renewalRaw=form.getRange('I28').getValue();
      output[22]=parseRenewalDate(renewalRaw instanceof Date
        ? Utilities.formatDate(renewalRaw,ss.getSpreadsheetTimeZone(),'yyyy-MM-dd')
        : renewalRaw);
    }
    [[2,'C6'],[13,'C18'],[19,'I22'],[25,'C30']].forEach(([col,address]) => {
      assertDropdownValue(list.getRange(6,col).getDataValidation(), input[address]);
    });
    if (rowNumber !== 6) {
      const template = list.getRange(6,2,1,27), target = list.getRange(rowNumber,2,1,27);
      template.copyTo(target,SpreadsheetApp.CopyPasteType.PASTE_FORMAT,false);
      template.copyTo(target,SpreadsheetApp.CopyPasteType.PASTE_DATA_VALIDATION,false);
    }
    [4,5,12,14,15,18,20,22,28].forEach(col => list.getRange(rowNumber,col).setNumberFormat('@'));
    list.getRange(rowNumber,10).setNumberFormat('0"일"');
    list.getRange(rowNumber,24).setNumberFormat('m"월 "d"일"');
    list.getRange(rowNumber,2,1,27).setValues([output]);
    committed = true;
    refreshCustomerListLayout(ss);
    refreshOperationalViews(ss);
    refreshCrmCalendar(ss);
    form.getRangeList(['B1','K4','K6','B36']).clearContent();
    form.getRangeList(Object.keys(CUSTOMER_FIELDS)).clearContent();
    ss.toast('고객 LIST에 등록했습니다.', '고객 등록');
  } catch (err) {
    const message = committed ? '저장은 완료됐으나 입력칸 정리에 실패했습니다. LIST를 먼저 확인하세요.' : err.message;
    // 먼저 발견한 오류 하나만 B1에 표시합니다.
    form.getRange('B1').setValue(message);
  } finally {
    lock.releaseLock();
  }
}

function parseCustomerPayday(value) {
  const text=String(value).trim();
  const day=Number(text);
  if(!/^[0-9]{1,2}$/.test(text)||!Number.isInteger(day)||day<1||day>31)
    throw new Error('카드 결제일은 1~31 사이의 숫자로 입력해 주세요.');
  return day;
}

function literalText(value) {
  // 사용자 텍스트가 수식으로 실행되지 않도록 처리합니다.
  return /^[=+\-@]/.test(value) ? "'"+value : value;
}

function setupCustomerButton() {
  const sheet = crmSheet_(SpreadsheetApp.getActiveSpreadsheet(),'고객 등록');
  const drawings = sheet.getDrawings();
  const assigned = drawings.filter(d => d.getOnAction() === 'registerCustomer');
  const button = assigned.length === 1 ? assigned[0] : drawings.length === 1 ? drawings[0] : null;
  if (!button) throw new Error('고객등록 그림이 하나인지 확인하거나 등록 그림에 registerCustomer를 할당해주세요.');
  button.setPosition(31,5,0,0).setWidth(220).setHeight(46).setOnAction('registerCustomer');
}

function formatResidentNumber(value) {
  const original = String(value);
  const s = original.trim();
  // 잘못된 문자나 길이를 조용히 삭제하지 않고 그대로 남깁니다.
  if (!/^\d{6}(?:-?\d{0,7})$/.test(s)) return original;
  const digits = s.replace('-','');
  return digits.slice(0,6)+'-'+digits.slice(6);
}

function birthFromResidentNumber(value) {
  const s = String(value).replace(/\D/g,'');
  if (![6,13].includes(s.length)) return '';
  const g = s[6];
  const century = s.length === 6
    ? (Number(s.slice(0,2)) <= new Date().getFullYear()%100 ? 2000 : 1900)
    : '3478'.includes(g) ? 2000 : '1256'.includes(g) ? 1900 : 1800;
  const year = century+Number(s.slice(0,2)), month = Number(s.slice(2,4)), day = Number(s.slice(4,6));
  const date = new Date(year,month-1,day);
  if (date.getFullYear() !== year || date.getMonth() !== month-1 || date.getDate() !== day || date > new Date()) return '';
  return s.slice(0,6);
}

function normalizeBirth(value) {
  const s = String(value).replace(/\D/g,'');
  if (s.length !== 6 && s.length !== 8) throw new Error('생년월일은 YYMMDD 6자리 또는 YYYYMMDD 8자리로 입력하세요.');
  const ymd = s.slice(-6), month = Number(ymd.slice(2,4)), day = Number(ymd.slice(4,6));
  const year = s.length === 8 ? Number(s.slice(0,4)) : 2000+Number(ymd.slice(0,2));
  const date = new Date(year,month-1,day);
  if (date.getFullYear() !== year || date.getMonth() !== month-1 || date.getDate() !== day)
    throw new Error('생년월일의 월·일을 확인하세요.');
  return ymd;
}

function parseRenewalDate(value) {
  let text;
  if(value instanceof Date){
    text=Utilities.formatDate(value,SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(),'yyyy-MM-dd');
  } else {
    text=String(value).trim();
  }
  let month,day,m;
  if(/^\d{8}$/.test(text)){month=Number(text.slice(4,6));day=Number(text.slice(6));}
  else if(/^\d{6}$/.test(text)){month=Number(text.slice(2,4));day=Number(text.slice(4));}
  else if(/^\d{3,4}$/.test(text)){month=Number(text.slice(0,-2));day=Number(text.slice(-2));}
  else if((m=/^(?:\d{2}|\d{4})\s*[-./]\s*(\d{1,2})\s*[-./]\s*(\d{1,2})$/.exec(text))){month=Number(m[1]);day=Number(m[2]);}
  else if((m=/^(?:\d{2,4}\s*년\s*)?(\d{1,2})\s*월\s*(\d{1,2})\s*일?$/.exec(text))){month=Number(m[1]);day=Number(m[2]);}
  else if((m=/^(\d{1,2})\s*[-./]\s*(\d{1,2})$/.exec(text))){month=Number(m[1]);day=Number(m[2]);}
  else throw new Error('갱신일의 월·일을 확인하세요. 예: 9월1일, 09-01, 20260901');
  const date=new Date(Date.UTC(2000,month-1,day));
  if(date.getUTCMonth()!==month-1||date.getUTCDate()!==day)throw new Error('갱신일의 월·일을 확인하세요.');
  return renewalDateFormula(month,day);
}

function renewalDateFormula(month,day) {
  return '=DATE(YEAR(TODAY())+IF(DATE(YEAR(TODAY()),'+month+','+day+')<TODAY(),1,0),'+month+','+day+')';
}

function nextRenewalSerial(month,day,todayText) {
  if(!todayText){
    const timezone=SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone();
    todayText=Utilities.formatDate(new Date(),timezone,'yyyy-MM-dd');
  }
  const parts=todayText.split('-').map(Number);
  const passed=month<parts[1] || (month===parts[1] && day<parts[2]);
  const year=parts[0]+(passed?1:0);
  const next=new Date(Date.UTC(year,month-1,day));
  if(next.getUTCMonth()!==month-1 || next.getUTCDate()!==day)
    throw new Error('다음 갱신 연도에 존재하지 않는 날짜입니다. 월·일을 확인하세요.');
  return Math.round((next.getTime()-Date.UTC(1899,11,30))/86400000);
}

function parseContractDate(value,todayText) {
  const today=(todayText||contractTodayText(SpreadsheetApp.getActiveSpreadsheet())).split('-').map(Number);
  let text=value instanceof Date?Utilities.formatDate(value,SpreadsheetApp.getActiveSpreadsheet().getSpreadsheetTimeZone(),'yyyy-MM-dd'):String(value==null?'':value).trim();
  let year=today[0],month,day,match;
  if(!text){month=today[1];day=today[2];}
  else if(/^\d{8}$/.test(text)){year=Number(text.slice(0,4));month=Number(text.slice(4,6));day=Number(text.slice(6));}
  else if(/^\d{6}$/.test(text)){year=2000+Number(text.slice(0,2));month=Number(text.slice(2,4));day=Number(text.slice(4));}
  else if(/^\d{3,4}$/.test(text)){month=Number(text.slice(0,-2));day=Number(text.slice(-2));}
  else if((match=/^(\d{2}|\d{4})\s*[-./,]\s*(\d{1,2})\s*[-./,]\s*(\d{1,2})$/.exec(text))){year=match[1].length===2?2000+Number(match[1]):Number(match[1]);month=Number(match[2]);day=Number(match[3]);}
  else if((match=/^(?:(\d{2}|\d{4})\s*년\s*)?(\d{1,2})\s*월\s*(\d{1,2})\s*일?$/.exec(text))){if(match[1])year=match[1].length===2?2000+Number(match[1]):Number(match[1]);month=Number(match[2]);day=Number(match[3]);}
  else if((match=/^(\d{1,2})\s*[-./,]\s*(\d{1,2})$/.exec(text))){month=Number(match[1]);day=Number(match[2]);}
  else throw new Error('계약일을 확인하세요. 예: 20260901, 260901, 9월1일, 09-01');
  const date=new Date(Date.UTC(year,month-1,day));
  if(year<1900||year>2199||date.getUTCFullYear()!==year||date.getUTCMonth()!==month-1||date.getUTCDate()!==day)throw new Error('계약일의 월·일을 확인하세요.');
  return Math.round((date.getTime()-Date.UTC(1899,11,30))/86400000);
}

function normalizeContractBirth(value) {
  let text=String(value==null?'':value).trim();
  if(!text)return '';
  let digits=text.replace(/\D/g,'');
  if(typeof value==='number'&&digits.length<6)digits=digits.padStart(6,'0');
  if(digits.length===13)digits=digits.slice(0,6);
  else if(digits.length===8)digits=digits.slice(-6);
  if(digits.length!==6||!birthFromResidentNumber(digits))throw new Error('생년월일을 6자리로 확인하세요. 예: 000101');
  return digits;
}

function parseContractMonths(value,label) {
  if(value===''||value==null)return 0;
  if(typeof value==='boolean')throw new Error('면책·감액은 숫자(개월 수)로 입력해 주세요');
  const text=String(value).trim().replace(/\s*개월$/,'');
  if(!/^\d+$/.test(text))throw new Error('면책·감액은 숫자(개월 수)로 입력해 주세요');
  const months=Number(text);
  if(!Number.isSafeInteger(months))throw new Error('면책·감액은 숫자(개월 수)로 입력해 주세요');
  return months;
}

function buildContractRecord(data,customers,todayText) {
  if(!String(data.ownerBirth||'').trim())throw new Error('계약자 생년월일을 입력하세요.');
  const owner=contractFindCustomer(customers,data.owner,data.ownerBirth,'계약자');
  const insured={name:String(data.insured||'').trim(),birth:normalizeContractBirth(data.insuredBirth),category:'',phone:''};
  const premiumText=String(data.premium==null?'':data.premium).replace(/[\s,원]/g,'');
  const premium=premiumText===''?'':Number(premiumText);
  if(premium!==''&&(!Number.isFinite(premium)||premium<0))throw new Error('보험료는 0 이상의 숫자로 입력하세요.');
  return ['',literalText(owner.category),parseContractDate(data.date,todayText),literalText(owner.name),owner.birth,literalText(owner.phone),literalText(insured.name),insured.birth,literalText(String(data.insurer||'').trim()),literalText(String(data.product||'').trim()),premium,parseContractMonths(data.immunity,'면책'),parseContractMonths(data.reduction,'감액'),literalText(String(data.memo||'').trim()),literalText(insured.category),literalText(insured.phone)];
}

function handleContractEdit(e) {
  if(!e||!e.range)return false;
  const sheet=e.range.getSheet(),name=crmSheetRole_(sheet);
  if(name!=='계약 입력'&&name!=='계약 목록')return false;
  if(name==='계약 입력'&&e.range.getA1Notation()==='F18'&&e.value==='TRUE'){try{registerContract(e.source);}finally{e.range.setValue(false);}return true;}
  const today=contractTodayText(e.source);
  const configs=name==='계약 입력'?[{row:4,col:3,date:true},{row:6,col:7,birth:true},{row:8,col:7,birth:true},{row:16,col:3,label:'면책'},{row:16,col:7,label:'감액'}]:[{row:null,col:3,date:true},{row:null,col:5,birth:true},{row:null,col:8,birth:true},{row:null,col:12,label:'면책'},{row:null,col:13,label:'감액'}];
  configs.forEach(config=>{
    if(e.range.getColumn()>config.col||e.range.getLastColumn()<config.col)return;
    if(config.row!==null&&(e.range.getRow()>config.row||e.range.getLastRow()<config.row))return;
    const start=config.row===null?Math.max(5,e.range.getRow()):config.row,end=config.row===null?e.range.getLastRow():config.row;
    if(start>end)return;
    const range=sheet.getRange(start,config.col,end-start+1,1),values=range.getValues(),formulas=range.getFormulas();
    values.forEach((row,index)=>{
      if(formulas[index][0]||row[0]===''||row[0]===null)return;
      const cell=sheet.getRange(start+index,config.col);
      try {const value=config.date?parseContractDate(row[0],today):config.birth?normalizeContractBirth(row[0]):parseContractMonths(row[0],config.label);cell.setNumberFormat(config.date?'yyyy-mm-dd':config.birth?'@':'0"개월"').setValue(value).clearNote();}
      catch(error){cell.setNote(error.message);if(name==='계약 입력')sheet.getRange('B1').setValue(error.message);else e.source.toast(error.message,'계약 입력',8);}
    });
  });
  if(name==='계약 목록'&&e.range.getLastRow()>=5){
    const start=e.range.getColumn(),end=e.range.getLastColumn();
    if([4,5,7,8].some(col=>start<=col&&col<=end)){
      const customers=contractCustomerRows(e.source);
      for(let row=Math.max(5,e.range.getRow());row<=e.range.getLastRow();row++){
        refreshContractRowCustomers(e.source,sheet,row,customers);
        if(sheet.getRange(row,4).getDisplayValue().trim()&&sheet.getRange(row,3).getValue()==='')sheet.getRange(row,3).setNumberFormat('yyyy-mm-dd').setValue(parseContractDate('',today));
      }
    }
  }
  if(name==='계약 목록'){refreshCustomerCard(e.source);refreshCrmCalendar(e.source);}
  return true;
}

function contractExpirySerial_(date,months) {
  if(date===''||date==null||months===''||months==null||Number(months)===0)return '';
  const count=parseContractMonths(months);
  const serial=typeof date==='number'?date:parseContractDate(date);
  const epoch=Date.UTC(1899,11,30),base=new Date(epoch+serial*86400000);
  const target=new Date(Date.UTC(base.getUTCFullYear(),base.getUTCMonth()+count,1));
  const last=new Date(Date.UTC(target.getUTCFullYear(),target.getUTCMonth()+1,0)).getUTCDate();
  target.setUTCDate(Math.min(base.getUTCDate(),last));
  return Math.round((target.getTime()-epoch)/86400000);
}

const CRM_CALENDAR_VIEWS=['캘린더','카드수납','자동차보험 갱신','면책·감액 종료'];
function parseCrmCalendarPeriod_(value) {
  const text=String(value==null?'':value).trim();let year,month,match;
  if((match=/^(\d{4})\s*년\s*(\d{1,2})\s*월$/.exec(text))){year=Number(match[1]);month=Number(match[2]);}
  else if((match=/^(\d{4})\s*[-./]\s*(\d{1,2})$/.exec(text))){year=Number(match[1]);month=Number(match[2]);}
  else if(/^\d{5,6}$/.test(text)){year=Number(text.slice(0,4));month=Number(text.slice(4));}
  else if(/^\d{8}$/.test(text)){year=Number(text.slice(0,4));month=Number(text.slice(4,6));}
  else if(/^\d{4}$/.test(text)){year=2000+Number(text.slice(0,2));month=Number(text.slice(2));}
  else throw new Error('월을 확인하세요. 예: 2026년 10월, 202610');
  if(year<1900||year>2199||month<1||month>12)throw new Error('월을 확인하세요. 예: 2026년 10월, 202610');
  return {year,month,label:year+'년'+month+'월'};
}
function crmCalendarSerial_(year,month,day) {return Math.round((Date.UTC(year,month-1,day)-Date.UTC(1899,11,30))/86400000);}
function crmCalendarDate_(serial) {return new Date(Date.UTC(1899,11,30)+Number(serial)*86400000);}
function crmCalendarInputDate_(value,ss) {
  if(value===''||value==null)return '';
  if(value instanceof Date){
    const formatted=Utilities.formatDate(value,ss.getSpreadsheetTimeZone(),'yyyy-MM-dd');
    value=Number(formatted.split('-')[0])>2199?Math.round((value.getTime()-Date.UTC(1899,11,30))/86400000):formatted;
  }
  if(typeof value==='number'&&value>0&&value<110000)return Math.floor(value);
  return parseContractDate(value,contractTodayText(ss));
}

function buildCrmCalendarModel_(year,month,today,customers,contracts,personal) {
  const first=crmCalendarSerial_(year,month,1),next=crmCalendarSerial_(year,month+1,1);
  const start=first-crmCalendarDate_(first).getUTCDay(),end=next-1;
  const events=[],texts={};
  const add=(date,view,type,text,name,birth)=>{if(date>=first&&date<=end&&(view===CRM_CALENDAR_VIEWS[0]||date>=today))events.push({date,view,type,text,name:name||'',birth:birth||''});};
  const clamp=(m,d)=>crmCalendarSerial_(year,m,Math.min(d,crmCalendarDate_(crmCalendarSerial_(year,m+1,1)-1).getUTCDate()));
  personal.forEach(row=>{const date=Number(row[0]),text=String(row[2]||'').trim();if(!date||!text||date<first||date>end)return;
    const displayed=(row[3]==='완료'?'✓ ':'')+text;
    texts[date]=(texts[date]?texts[date]+'\n':'')+displayed;
    add(date,CRM_CALENDAR_VIEWS[0],'개인일정',displayed);
  });
  const todayYear=crmCalendarDate_(today).getUTCFullYear();
  customers.forEach(row=>{
    const name=String(row[1]||'').trim();if(!name)return;
    let rrn=String(row[2]||'').replace(/\D/g,'');if(rrn.length<6&&rrn)rrn=rrn.padStart(6,'0');
    const birth=birthFromResidentNumber(rrn);
    if(birth){const yy=Number(birth.slice(0,2)),bm=Number(birth.slice(2,4)),bd=Number(birth.slice(4));
      const century=rrn.length===13?('3478'.includes(rrn[6])?2000:'1256'.includes(rrn[6])?1900:1800):(yy<=todayYear%100?2000:1900);
      if(year>=century+yy)add(clamp(bm,bd),CRM_CALENDAR_VIEWS[0],'고객생일',name+' 생일');
    }
    const payday=Number(String(row[8]||'').replace(/일$/,''));
    const lastDay=crmCalendarDate_(end).getUTCDate();
    if(Number.isInteger(payday)&&payday>=1&&payday<=lastDay)add(crmCalendarSerial_(year,month,payday),CRM_CALENDAR_VIEWS[1],'카드수납',name+' '+birth+(row[11]?' · '+row[11]:''),name,birth);
    const renewal=Number(row[22]);if(renewal>0){const date=crmCalendarDate_(renewal),day=clamp(date.getUTCMonth()+1,date.getUTCDate());add(day,CRM_CALENDAR_VIEWS[2],'자동차보험 갱신',name+' '+birth+(row[20]?' · '+row[20]:''),name,birth);}
  });
  contracts.forEach(row=>{
    const date=Number(row[2]),name=String(row[3]||'').trim();if(!date||!name)return;
    [[11,'면책 종료'],[12,'감액 종료']].forEach(([col,type])=>{const months=Number(row[col]);if(Number.isInteger(months)&&months>0){const expiry=contractExpirySerial_(date,months);const birth=String(row[4]||'').replace(/\D/g,'').padStart(6,'0');add(expiry,CRM_CALENDAR_VIEWS[3],type,name+' '+birth+(row[9]?' · '+row[9]:''),name,birth);}});
  });
  const weeks=Array.from({length:6},(_,index)=>{const a=Math.max(first,start+index*7),b=Math.min(end,start+index*7+6);return {start:a,end:b,valid:a<=b};});
  events.sort((a,b)=>a.date-b.date||a.type.localeCompare(b.type,'ko'));
  return {first,end,start,today,weeks,events,personal:texts};
}

function refreshCrmCalendar(source,dailyOnly) {
  const lock=LockService.getDocumentLock();if(!lock.tryLock(100))return;
  try{return refreshCrmCalendarUnlocked_(source,dailyOnly);}finally{lock.releaseLock();}
}
function refreshCrmCalendarUnlocked_(source,dailyOnly) {
  const ss=source&&typeof source.getSheetByName==='function'?source:SpreadsheetApp.getActiveSpreadsheet();
  const sheet=crmSheet_(ss,'캘린더');if(!sheet)return;
  ensureCrmCalendarStorage_(sheet);
  const todayText=contractTodayText(ss);if(dailyOnly&&sheet.getRange('AK1').getDisplayValue()===todayText)return;
  const current=todayText.split('-').map(Number),year=Number(sheet.getRange('D2').getValue())||current[0],month=Number(sheet.getRange('E2').getValue())||current[1];
  if(!Number.isInteger(year)||year<1900||year>2199||!Number.isInteger(month)||month<1||month>12){sheet.getRange('B1').setValue('연도와 월을 숫자로 확인하세요.');return;}
  sheet.getRange('D2').setNumberFormat('0"년"').setValue(year);sheet.getRange('E2').setNumberFormat('0"월"').setValue(month);
  const view=CRM_CALENDAR_VIEWS[0],key=year+'-'+month;
  const list=crmSheet_(ss,'고객 LIST'),contractsSheet=crmSheet_(ss,'계약 목록');
  const customers=list?list.getRange(6,2,Math.max(1,list.getLastRow()-5),27).getValues():[];
  customers.forEach(row=>{if(row[22] instanceof Date)row[22]=crmCalendarInputDate_(row[22],ss);});
  const contracts=contractsSheet?contractsSheet.getRange(5,1,Math.max(1,contractsSheet.getLastRow()-4),14).getValues():[];
  contracts.forEach(row=>{if(row[2] instanceof Date)row[2]=crmCalendarInputDate_(row[2],ss);});
  const personal=sheet.getRange(5,31,Math.max(1,sheet.getLastRow()-4),4).getValues();
  personal.forEach(row=>{try{row[0]=crmCalendarInputDate_(row[0],ss);}catch(error){row[0]='';}});
  const model=buildCrmCalendarModel_(year,month,parseContractDate(todayText,todayText),customers,contracts,personal);
  model.holidays=crmCalendarHolidays_(sheet);
  renderCrmCalendar_(sheet,model,view);
  if(!Object.keys(model.holidays).some(date=>crmCalendarDate_(Number(date)).getUTCFullYear()===year)){
    const prefix=view===CRM_CALENDAR_VIEWS[0]?'':'오늘 이후 일정만 표시합니다. ';
    sheet.getRange('B1').setValue(prefix+'해당 연도의 공휴일 정보가 아직 없습니다.');
  }
  sheet.getRange('AK1').setNumberFormat('@').setValue(todayText);sheet.getRange('AK2').setNumberFormat('@').setValue(key);
}


function ensureCrmCalendarStorage_(sheet) {
  if(sheet.getMaxColumns()<44)sheet.insertColumnsAfter(sheet.getMaxColumns(),44-sheet.getMaxColumns());
  if(sheet.getRange('AK5').getValue()==='v2')return;
  const old=sheet.getRange(28,10,sheet.getMaxRows()-27,4).getValues().filter(r=>r[0]&&r[2]);
  if(old.length)sheet.getRange(5,31,old.length,4).setValues(old);
  const holidays=sheet.getRange(4,15,sheet.getMaxRows()-3,2).getValues().filter(r=>r[0]&&r[1]);
  if(holidays.length)sheet.getRange(5,35,holidays.length,2).setValues(holidays);
  sheet.getRange('AK5').setValue('v2');
}
function renderCrmCalendar_(sheet,model) {
  sheet.getRange('B1').clearContent();
  sheet.getRange('B2').clearDataValidations().setValue('캘린더');
  sheet.getRange('D2:E2').clearDataValidations();
  sheet.getRange('G4:H5').clearContent().clearDataValidations();
  sheet.showColumns(10,6);sheet.hideColumns(16,sheet.getMaxColumns()-15);
  sheet.setColumnWidth(10,100);sheet.setColumnWidth(11,100);sheet.setColumnWidth(12,85);sheet.setColumnWidth(13,125);sheet.setColumnWidth(14,260);sheet.setColumnWidth(15,75);
  sheet.getRange('J2:O2').breakApart().merge().setValue('날짜별 일정');sheet.getRange('J3').setValue('조회 날짜');

  // 예전 3행 달력의 유효성 검사만 제거합니다. 달력 틀과 테두리는 유지합니다.
  sheet.getRange('B4:H34').clearDataValidations().clearNote();
  sheet.getRange('B4:H4').setValues([['일','월','화','수','목','금','토']]).setFontWeight('bold').setHorizontalAlignment('center');  const colors=Array.from({length:30},()=>Array(7).fill('#202124'));
  const calendarDateTargets={},values=Array.from({length:30},()=>Array(7).fill(''));
  for(let week=0;week<6;week++)for(let day=0;day<7;day++){
    const date=model.start+week*7+day,row=5+week*5,col=2+day,inMonth=date>=model.first&&date<=model.end;
    values[week*5][day]=crmCalendarDate_(date).getUTCDate();
    colors[week*5][day]=!inMonth?'#b7b7b7':day===0||(model.holidays||{})[date]?'#d93025':day===6?'#1a73e8':'#202124';
    if(!inMonth)continue;
    const lines=String(model.personal[date]||'').split('\n');
    for(let k=0;k<4;k++){
      values[week*5+1+k][day]=literalText(k===3?lines.slice(k).join('\n'):(lines[k]||''));
      calendarDateTargets[(row+1+k)+':'+col]=date;
    }
    calendarDateTargets[row+':'+col]=date;
  }
  sheet.getRange('B5:H34').setNumberFormat('@').setValues(values).setFontColors(colors).setFontWeight('normal');
  for(let week=0;week<6;week++){
    const row=5+week*5;    sheet.getRange(row,2,1,7).setNumberFormat('0').setHorizontalAlignment('right').setFontWeight('bold');
  }

  PropertiesService.getDocumentProperties().setProperty('CRM_CALENDAR_DATE_MAP',JSON.stringify(calendarDateTargets));
  sheet.getRange(5,38,sheet.getMaxRows()-4,7).clearContent();
  if(model.events.length)sheet.getRange(5,38,model.events.length,7).setValues(model.events.map(e=>[e.date,e.type,literalText(e.name),literalText(e.birth),literalText(e.text),e.view,'']));
  let selected;try{selected=crmCalendarInputDate_(sheet.getRange('K3').getValue(),sheet.getParent());}catch(error){}
  if(!selected||selected<model.first||selected>model.end)selected=model.today>=model.first&&model.today<=model.end?model.today:model.first;
  sheet.getRange('K3').setValue(selected).setNumberFormat('yyyy-mm-dd');
  renderCrmCalendarDetails_(sheet,selected);
}
function renderCrmCalendarDetails_(sheet,date) {
  const rows=sheet.getRange(5,38,Math.max(1,sheet.getLastRow()-4),7).getValues().filter(r=>Number(r[0])===date);
  sheet.getRange('J4:O4').setValues([['날짜','고객이름','생년월일','구분','내용','고객카드']]);

  if(rows.length){
    sheet.getRange(5,10,rows.length,6).setValues(rows.map(r=>[date,literalText(r[2]),literalText(r[3]),r[1],literalText(r[4]),false]));
    sheet.getRange(5,10,rows.length,1).setNumberFormat('m월 d일');
    sheet.getRange(5,12,rows.length,1).setNumberFormat('@');
    rows.forEach((r,i)=>{if(r[2])sheet.getRange(i+5,15).insertCheckboxes();});

}
function saveCalendarPersonalDay_(sheet,date,text) {
  const data=sheet.getRange(5,31,sheet.getMaxRows()-4,4).getValues();let empty=-1;const matches=[];
  data.forEach((r,i)=>{if(!r[0]&&empty<0)empty=i;try{if(crmCalendarInputDate_(r[0],sheet.getParent())===date)matches.push(i);}catch(error){}});
  matches.forEach(i=>sheet.getRange(i+5,31,1,4).clearContent());
  if(!String(text||'').trim())return;
  const dest=matches.length?matches[0]:empty;
  if(dest<0)throw new Error('개인일정 저장 공간이 부족합니다.');
  sheet.getRange(dest+5,31,1,4).setValues([[date,'개인일정',literalText(String(text).trim()),'예정']]);
}

function onSelectionChange(e) {
  if(!e||!e.range||crmSheetRole_(e.range.getSheet())!=='캘린더'||e.range.getNumRows()!==1||e.range.getNumColumns()!==1)return;
  const sheet=e.range.getSheet(),row=e.range.getRow(),col=e.range.getColumn();
  const selectedDateTargets=JSON.parse(PropertiesService.getDocumentProperties().getProperty('CRM_CALENDAR_DATE_MAP')||'{}');

  if(date){sheet.getRange('K3').setValue(date).setNumberFormat('yyyy-mm-dd');renderCrmCalendarDetails_(sheet,date);}

}
function handleCrmCalendarEdit(e) {
  if(!e||!e.range||crmSheetRole_(e.range.getSheet())!=='캘린더')return false;
  const sheet=e.range.getSheet(),ss=e.source;
  try{
    if(e.range.getColumn()===15&&e.range.getRow()>=5&&e.value==='TRUE'){
      const r=sheet.getRange(e.range.getRow(),11,1,2).getDisplayValues()[0];e.range.setValue(false);openCalendarCustomer_(ss,r[0],r[1]);return true;
    }
    if(e.range.getA1Notation()==='K3'){const date=crmCalendarInputDate_(e.range.getValue(),ss);e.range.setValue(date).setNumberFormat('yyyy-mm-dd');renderCrmCalendarDetails_(sheet,date);return true;}
    const targets=JSON.parse(PropertiesService.getDocumentProperties().getProperty('CRM_CALENDAR_DATE_MAP')||'{}');
    for(let w=0;w<6;w++){
      const header=5+w*5,first=header+1,last=header+4;
      if(e.range.getRow()>last||e.range.getLastRow()<first)continue;
      for(let col=Math.max(2,e.range.getColumn());col<=Math.min(8,e.range.getLastColumn());col++){
        const date=targets[header+':'+col];if(!date)continue;
        const text=sheet.getRange(first,col,4,1).getDisplayValues().map(r=>r[0]).filter(t=>t.trim()).join('\n');
        saveCalendarPersonalDay_(sheet,date,text);
      }
    }
    refreshCrmCalendar(ss);
  }catch(error){sheet.getRange('B1').setValue(error.message);}
  return true;
}


// 한국천문연구원의 2026 월력요항을 기본 자료로 사용하고 Google 공개 공휴일 달력으로 갱신합니다.
const CRM_KOREAN_HOLIDAYS_2026=[
 ['20260101','신정'],['20260216','설날 연휴'],['20260217','설날'],['20260218','설날 연휴'],
 ['20260301','삼일절'],['20260302','대체공휴일'],['20260501','노동절'],['20260505','어린이날'],
 ['20260524','부처님오신날'],['20260525','대체공휴일'],['20260603','전국동시지방선거'],['20260606','현충일'],
 ['20260717','제헌절'],['20260815','광복절'],['20260817','대체공휴일'],['20260924','추석 연휴'],['20260925','추석'],['20260926','추석 연휴'],
 ['20261003','개천절'],['20261005','대체공휴일'],['20261009','한글날'],['20261225','성탄절']
];
function parseCrmHolidayIcs_(text,official) {
  const unfolded=String(text).replace(/\r?\n[ \t]/g,'');
  const events=[],blocks=unfolded.split('BEGIN:VEVENT').slice(1);
  blocks.forEach(block=>{
    const date=/^DTSTART(?:;[^:\r\n]*)?:(\d{8})/m.exec(block),name=/^SUMMARY(?:;[^:\r\n]*)?:(.+)$/m.exec(block);
    if(!date||!name)return;
    const label=name[1].trim().replace(/\\n/g,' ').replace(/\\,/g,',').replace(/\\;/g,';');
    const newHoliday=Number(date[1].slice(0,4))>=2026&&/노동절|근로자의날|근로자의 날|제헌절/.test(label);
    if(!official&&!newHoliday&&!/신정|새해|설날|설연휴|삼일절|3[·.\s]*1절|어린이날|부처님|석가탄신일|현충일|광복절|추석|개천절|한글날|성탄절|기독탄신일|크리스마스|대체.*공휴일|임시.*공휴일|선거/.test(label))return;
    events.push([crmCalendarSerial_(Number(date[1].slice(0,4)),Number(date[1].slice(4,6)),Number(date[1].slice(6))),label]);
  });
  return events;
}
function crmCalendarHolidays_(sheet) {
  const result={};
  CRM_KOREAN_HOLIDAYS_2026.forEach(([date,label])=>{result[crmCalendarSerial_(Number(date.slice(0,4)),Number(date.slice(4,6)),Number(date.slice(6)))]=label;});
  const rows=sheet.getRange(5,35,Math.max(1,sheet.getLastRow()-4),2).getValues();
  rows.forEach(row=>{if(row[0]&&row[1]){
    const date=row[0] instanceof Date?crmCalendarInputDate_(row[0],sheet.getParent()):Number(row[0]);
    result[date]=String(row[1]);
  }});
  return result;
}
function updateCrmHolidayData_(ss) {
  const sheet=crmSheet_(ss,'캘린더');if(!sheet)return;
  let events=[];
  for(const official of [true,false]){
    const calendar=official?'ko.south_korea.official':'ko.south_korea';
    try{
      const response=UrlFetchApp.fetch('https://calendar.google.com/calendar/ical/'+calendar+'%23holiday%40group.v.calendar.google.com/public/basic.ics',{muteHttpExceptions:true});
      if(response.getResponseCode()===200){events=parseCrmHolidayIcs_(response.getContentText(),official);if(events.length)break;}
    }catch(error){}
  }
  if(!events.length)throw new Error('공휴일 자료를 불러오지 못했습니다. 2026년 기본 공휴일 자료는 유지됩니다.');
  const unique={};events.forEach(([date,name])=>{unique[date]=name;});
  const rows=Object.keys(unique).map(Number).sort((a,b)=>a-b).map(date=>[date,unique[date]]);
  if(rows.length+4>sheet.getMaxRows())sheet.insertRowsAfter(sheet.getMaxRows(),rows.length+4-sheet.getMaxRows());
  sheet.getRange(5,35,sheet.getMaxRows()-4,2).clearContent();
  sheet.getRange(5,35,rows.length,2).setValues(rows);
  sheet.getRange('AK3').setNumberFormat('@').setValue(contractTodayText(ss));
}

  const ss=SpreadsheetApp.getActiveSpreadsheet();
  ensureCrmCalendarStorage_(crmSheet_(ss,'캘린더'));
  try{updateCrmHolidayData_(ss);}catch(error){ss.toast(error.message,'공휴일 자료');}
  refreshCrmCalendar(ss);
  setupViewAutoRefresh();
}

function refreshCrmHolidayDataIfNeeded_(ss) {
  const sheet=crmSheet_(ss,'캘린더');if(!sheet)return;
  const todayText=contractTodayText(ss),last=sheet.getRange('AK3').getDisplayValue(),attempt=sheet.getRange('AK4').getDisplayValue();
  if(attempt===todayText)return;
  if(last&&parseContractDate(todayText)-parseContractDate(last)<30)return;
  sheet.getRange('AK4').setNumberFormat('@').setValue(todayText);
  try{updateCrmHolidayData_(ss);}catch(error){}
}
